   :param client: An authenticated API Client instance
   :type client: ApiClient

   :type query: str

   :type total: int

   :type phrase: str
   :param partial: Whether a statement hit its time limit and some hits are missing.
   :type partial: bool
   :param truncated: Whether the user has more sessions than the search covered.
   :type truncated: bool
   :param linesCapped: Whether transcript lines were searched for the newest sessions only; titles and summaries covered them all.
   :type linesCapped: bool
   :param queryTerms: The search terms the server used.
   :type queryTerms: AudioListenerSearchTerm
   :param results: The list of matching transcript search hits.
   :type results: AudioListenerTranscriptSearchResult

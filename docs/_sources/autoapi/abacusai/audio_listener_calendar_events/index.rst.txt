   :param client: An authenticated API Client instance
   :type client: ApiClient
   :param providers: Each calendar provider (google, outlook): whether it is connected, its account, and an error if it could not be read.
   :type providers: list
   :param events: The meetings in the window, by start time, from every connected calendar.
   :type events: list
   :param startMs: The start of the window read, epoch milliseconds.
   :type startMs: int
   :param endMs: The end of the window read, epoch milliseconds.
   :type endMs: int
